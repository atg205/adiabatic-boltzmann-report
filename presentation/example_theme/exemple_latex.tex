\section{The basics: Tolerant testing, stabilizer states, Norm required to understand the article}

\begin{frame}{Classical and Quantum Property Testing}

\textbf{Classical Property Testing}

\begin{itemize}
    \item Property $P \subseteq \{0,1\}^{\{0,1\}^n}$ (set of Boolean functions)
    \item Oracle access to an unknown function $g$
    \item Promise problem:
    \[
        \text{either } g \in P
        \quad \text{or} \quad
        \mathrm{dist}(g,P) \ge \varepsilon
    \]
    where
    \[
        \mathrm{dist}(g,P)
        =
        \min_{f \in P} \Pr_x[g(x)\neq f(x)]
    \]
    \item Goal: decide which case holds using few queries
\end{itemize}

\vspace{0.3cm}

\textbf{Quantum Property Testing}

\begin{itemize}
    \item Property $P$ = set of $n$-qubit quantum states
    \item Access to independent copies of an unknown state $|\psi\rangle$
    \item Promise problem:
    \[
        \text{either } |\psi\rangle \in P
        \quad \text{or} \quad
        \mathrm{dist}(|\psi\rangle,P) \ge \varepsilon
    \]
    where
    \[
        \mathrm{dist}(|\psi\rangle,P)
        =
        \min_{|\phi\rangle \in P}
        \sqrt{1 - |\langle \psi | \phi \rangle|^2}
    \]
    \item Goal: decide using few copies and joint measurements
\end{itemize}

\textbf{Why Tolerant Testing is Harder}
\begin{itemize}
    \item Direct (exact) testing: check stabilizer generators $\rightarrow$ single-copy measurement suffices
    \item Tolerant testing: must estimate correlations / norms with high precision
    \item Noise or small deviations can hide stabilizer structure
    \item Requires \textit{statistical sampling, many copies, careful analysis} → higher complexity
\end{itemize}

\end{frame}


\begin{frame} 
\frametitle{There Is No Largest Prime Number} 
\framesubtitle{The proof uses \textit{reductio ad absurdum}.} 
\begin{theorem}
	There is no \textbf{largest} prime number. \end{theorem} 
\begin{enumerate} 
	\item<1-| alert@1> Suppose $p$ were the largest prime number. 
	\item<2-> Let $q$ be the product of the first $p$ numbers. 
	\item<3-> Then $q+1$ is not divisible by any of them. 
	\item<1-> But $q + 1$ is greater than $1$, thus divisible by some prime
	number not in the first $p$ numbers.
\end{enumerate}
\end{frame}
